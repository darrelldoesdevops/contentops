```rust
fn wrap_title_lines(text: &str, font_size: u32, max_width: u32) -> Vec<String> {
    let avg_char_px = font_size as f64 * 0.55;
    let max_chars = (max_width as f64 / avg_char_px).floor() as usize;

    let mut result = Vec::new();
    for input_line in text.split('\n') {
        let words: Vec<&str> = input_line.trim().split_whitespace().collect();
        if words.is_empty() {
            continue;
        }
        let mut current_line = String::new();

        for word in words {
            let candidate = if current_line.is_empty() {
                word.to_string()
            } else {
                format!("{} {}", current_line, word)
            };

            if candidate.len() <= max_chars || current_line.is_empty() {
                current_line = candidate;
            } else {
                result.push(current_line.clone());
                current_line = word.to_string();
            }
        }
        if !current_line.is_empty() {
            result.push(current_line);
        }
    }
    result
}
```

Then in `build_title_filter()`, replace the existing line-splitting logic:

The 0.55 multiplier is conservative for Impact font uppercase text -- it slightly overestimates character width, causing earlier wrapping which is the safe direction.

User decision: wrap to next line when too wide (not shrink font). This implementation honors that.
